# База: лёгкий Debian Slim с Python 3.11.9
FROM python:3.11.9-slim

# Можно переключать CUDA-линию PyTorch при сборке: cu124 (по умолчанию) или cu121
ARG TORCH_CUDA_TAG="cu124"

ENV PYTHONDONTWRITEBYTECODE=1 \
    PYTHONUNBUFFERED=1 \
    PIP_NO_CACHE_DIR=1 \
    DEBIAN_FRONTEND=noninteractive

# Минимум системных пакетов + ffmpeg
RUN apt-get update && apt-get install -y --no-install-recommends \
    ffmpeg \
    ca-certificates \
    git \
    curl \
    && rm -rf /var/lib/apt/lists/*

# Обновим инструменты установки
RUN python -m pip install --upgrade pip setuptools wheel

# Установка PyTorch с CUDA 12.4 (cu124) из официального индекса + базовые научные пакеты
# При желании зафиксируйте версию torch (например, ==2.5.*)
RUN python -m pip install --no-cache-dir --index-url https://download.pytorch.org/whl/${TORCH_CUDA_TAG} \
      torch torchvision torchaudio && \
    python -m pip install --no-cache-dir \
      numpy scipy tqdm pyyaml rich

# Нерутовый пользователь
RUN useradd -ms /bin/bash appuser
USER appuser
WORKDIR /app

# Если есть зависимости/код — раскомментируйте:
# COPY requirements.txt ./
# RUN python -m pip install --no-cache-dir -r requirements.txt
# COPY . .

# По умолчанию просто показать версию Python (замените на свою команду приложения)
CMD ["python", "-V"]
